1️⃣ SaveToFile(level, "name")
   ┌─ KIEDY: Gracz chce zapisać swoją grę
   ├─ CO ROBI: Konwertuje level do JSON, zapisuje na dysk
   └─ WYNIK: Plik .lvl w SavedLevels/
   
   LevelSerializer.SaveToFile(myLevel, "checkpoint1");


2️⃣ LoadFromFile("name.lvl")
   ┌─ KIEDY: Gracz chce zagrać wszcześniejszy poziom
   ├─ CO ROBI: Wczytuje JSON z dysku, deserializuje do LevelData
   └─ WYNIK: LevelData gotowe do gry
   
   LevelData loaded = LevelSerializer.LoadFromFile("checkpoint1.lvl");


3️⃣ GetSaveDirectory()
   ┌─ KIEDY: Chcesz wiedzieć gdzie są pliki
   ├─ CO ROBI: Zwraca ścieżkę do folderu SavedLevels
   └─ WYNIK: "/Users/.../SavedLevels"
   
   string path = LevelSerializer.GetSaveDirectory();


4️⃣ GetSavedLevels()
   ┌─ KIEDY: Chcesz wyświetlić listę do załadowania
   ├─ CO ROBI: Zwraca tablicę wszystkich .lvl plików
   └─ WYNIK: string[] z ścieżkami
   
   var allLevels = LevelSerializer.GetSavedLevels();
   foreach (var file in allLevels) { Debug.Log(file); }


5️⃣ DeleteSavedLevel(path)
   ┌─ KIEDY: Gracz chce usunąć zapis
   ├─ CO ROBI: Usuwa plik z dysku
   └─ WYNIK: Plik gone
   
   LevelSerializer.DeleteSavedLevel("old_save.lvl");


6️⃣ ExportToCSV(level)
   ┌─ KIEDY: Designer chce analizować strukturę w Excelu
   ├─ CO ROBI: Konwertuje pokoje na tabelę CSV
   └─ WYNIK: String z danymi CSV
   
   string data = LevelSerializer.ExportToCSV(level);
   File.WriteAllText("analiza.csv", data);


7️⃣ ExportToGraphML(level)
   ┌─ KIEDY: Designer chce wizualną mapę pokoi
   ├─ CO ROBI: Konwertuje na format GraphML
   └─ WYNIK: XML plik do wizualizacji
   
   string xml = LevelSerializer.ExportToGraphML(level);
   File.WriteAllText("mapa.graphml", xml);

SZABLON 1: Podstawowy save/load

void SaveMyLevel(LevelData level)
{
    LevelSerializer.SaveToFile(level, "my_save");
}

LevelData LoadMyLevel()
{
    return LevelSerializer.LoadFromFile("my_save.lvl");
}


SZABLON 2: Z obsługą błędów
────────────────────────────
void SaveWithError(LevelData level)
{
    try
    {
        LevelSerializer.SaveToFile(level, "save");
        Debug.Log("✓ Zapisano");
    }
    catch (System.Exception ex)
    {
        Debug.LogError("✗ Błąd: " + ex.Message);
    }
}


SZABLON 3: Export
─────────────────
void ExportAll(LevelData level)
{
    // CSV
    File.WriteAllText("data.csv", LevelSerializer.ExportToCSV(level));
    
    // GraphML
    File.WriteAllText("map.graphml", LevelSerializer.ExportToGraphML(level));
    
    Debug.Log("✓ Wyeksportowano!");
}

void ShowSaveLoadMenu()
{
    var saves = LevelSerializer.GetSavedLevels();
    
    foreach (var file in saves)
    {
        string name = Path.GetFileNameWithoutExtension(file);
        Debug.Log($"→ {name}");
    }
}
